- ✅ `root_squash`, CIDR restriction, এবং secure defaults

- ✅ ইউজার-ডিফাইন্ড শেয়ার ডিরেক্টরি (যেমন: `kubernetes`, `wordpress`, `openstack`)
- ✅ Idempotent & production-ready

## 📄 1. `config/nfs.conf` — কেন্দ্রীয় কনফিগ

```bash
# config/nfs.conf
NFS_SHARE_BASE="/nfs-share"
NFS_SUBDIRS=("kubernetes" "docker/wordpress-data" "openstack")
NFS_NETWORK_CIDR="192.168.0.0/24"
NFS_EXPORT_OPTIONS="rw,sync,root_squash,no_subtree_check"
NFS_OWNER_USER="nobody"
NFS_OWNER_GROUP="nogroup"
```

> ⚠️ এটি **bash-compatible** — সরাসরি `source config/nfs.conf` করা যাবে।

## 📜 2. `scripts/install-nfs-server.sh`

```bash
#!/bin/bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" &>/dev/null && pwd)"
CONFIG_FILE="$SCRIPT_DIR/../config/nfs.conf"

if [[ ! -f "$CONFIG_FILE" ]]; then
  echo "❌ Config file not found: $CONFIG_FILE" >&2
  exit 1
fi

source "$CONFIG_FILE"

echo "🔧 Creating base share directory..."
mkdir -p "$NFS_SHARE_BASE"

echo "📂 Creating subdirectories..."
for dir in "${NFS_SUBDIRS[@]}"; do
  mkdir -p "$NFS_SHARE_BASE/$dir"
done

echo "👤 Setting ownership and permissions..."
chown -R "$NFS_OWNER_USER":"$NFS_OWNER_GROUP" "$NFS_SHARE_BASE"
chmod -R 755 "$NFS_SHARE_BASE"

echo "📦 Installing NFS server..."
apt update -y
apt install -y nfs-kernel-server

echo "📝 Generating /etc/exports..."
cat > /etc/exports <<EOF
# Auto-generated by nfs-server setup
$NFS_SHARE_BASE $NFS_NETWORK_CIDR($NFS_EXPORT_OPTIONS)
EOF

for dir in "${NFS_SUBDIRS[@]}"; do
  echo "$NFS_SHARE_BASE/$dir $NFS_NETWORK_CIDR($NFS_EXPORT_OPTIONS)" >> /etc/exports
done

echo "🔄 Applying exports..."
exportfs -ra
systemctl restart nfs-kernel-server
systemctl enable nfs-kernel-server

# Enable UFW for NFSv4 (if active)
if command -v ufw &> /dev/null && ufw status | grep -q "Status: active"; then
  echo "🧱 Configuring UFW..."
  ufw allow from "$NFS_NETWORK_CIDR" to any port 2049 proto tcp
fi

echo "✅ NFS Server setup complete!"
echo "📁 Base share: $NFS_SHARE_BASE"
echo "🌐 Accessible from: $NFS_NETWORK_CIDR"
```

## 💻 3. `scripts/install-nfs-client.sh`

```bash
#!/bin/bash
set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" &>/dev/null && pwd)"
CONFIG_FILE="$SCRIPT_DIR/../config/nfs.conf"

source "$CONFIG_FILE"

# Prompt or read from env
read -p "Enter NFS Server IP (e.g., 192.168.0.35): " NFS_SERVER_IP
read -p "Enter subdir to mount (e.g., kubernetes): " SUBDIR

MOUNT_POINT="/mnt/nfs-$SUBDIR"

echo "📦 Installing NFS client..."
apt update -y
apt install -y nfs-common

echo "📁 Creating mount point: $MOUNT_POINT"
mkdir -p "$MOUNT_POINT"

echo "🔗 Mounting NFS share (NFSv4)..."
mount -t nfs4 "$NFS_SERVER_IP:/$SUBDIR" "$MOUNT_POINT"

# Add to fstab
if ! grep -q "$NFS_SERVER_IP" /etc/fstab; then
  echo "$NFS_SERVER_IP:/$SUBDIR $MOUNT_POINT nfs4 defaults 0 0" >> /etc/fstab
fi

echo "✅ Mounted $NFS_SERVER_IP:/$SUBDIR → $MOUNT_POINT"
```

> 💡 NFSv4-এ `/nfs-share` ভার্চুয়াল রুট, তাই `:/kubernetes` লিখলেই হয়।

## 🧩 4. Ansible Automation (পূর্ণ অটোমেশন)

### `ansible/inventory.ini`
```ini
[nfs-servers]
nfs-server-01 ansible_host=192.168.0.35

[nfs-clients]
client-01 ansible_host=192.168.0.40
client-02 ansible_host=192.168.0.41
```

### `ansible/group_vars/all.yml`
```yaml
nfs_share_base: "/nfs-share"
nfs_subdirs:
  - "kubernetes"
  - "docker/wordpress-data"
  - "openstack"
nfs_network_cidr: "192.168.0.0/24"
nfs_export_options: "rw,sync,root_squash,no_subtree_check"
nfs_owner_user: "nobody"
nfs_owner_group: "nogroup"
```

### `ansible/templates/exports.j2`
```jinja2
# Managed by Ansible
{{ nfs_share_base }} {{ nfs_network_cidr }}({{ nfs_export_options }})
{% for subdir in nfs_subdirs %}
{{ nfs_share_base }}/{{ subdir }} {{ nfs_network_cidr }}({{ nfs_export_options }})
{% endfor %}
```

### `ansible/nfs-server.yml`
```yaml
---
- name: Setup NFS Server
  hosts: nfs-servers
  become: yes
  tasks:
    - name: Install NFS kernel server
      apt:
        name: nfs-kernel-server
        state: present
        update_cache: yes

    - name: Create base share directory
      file:
        path: "{{ nfs_share_base }}"
        state: directory
        owner: "{{ nfs_owner_user }}"
        group: "{{ nfs_owner_group }}"
        mode: '0755'

    - name: Create subdirectories
      file:
        path: "{{ nfs_share_base }}/{{ item }}"
        state: directory
        owner: "{{ nfs_owner_user }}"
        group: "{{ nfs_owner_group }}"
        mode: '0755'
      loop: "{{ nfs_subdirs }}"

    - name: Configure /etc/exports
      template:
        src: exports.j2
        dest: /etc/exports
        owner: root
        group: root
        mode: '0644'

    - name: Reload exports
      command: exportfs -ra
      notify: restart nfs

    - name: Enable UFW for NFSv4
      ufw:
        rule: allow
        from_ip: "{{ nfs_network_cidr }}"
        port: "2049"
        proto: tcp
      when: ansible_distribution == "Ubuntu"

  handlers:
    - name: restart nfs
      systemd:
        name: nfs-kernel-server
        state: restarted
        enabled: yes
```

### `ansible/nfs-client.yml`
```yaml
---
- name: Setup NFS Clients
  hosts: nfs-clients
  become: yes
  vars:
    nfs_server_ip: "192.168.0.35"
  tasks:
    - name: Install nfs-common
      apt:
        name: nfs-common
        state: present

    - name: Create mount points and mount shares
      mount:
        path: "/mnt/nfs-{{ item }}"
        src: "{{ nfs_server_ip }}:/{{ item }}"
        fstype: nfs4
        opts: defaults
        state: mounted
      loop: "{{ nfs_subdirs }}"
```

A fully automated, secure, and parameterized NFS setup for DevOps & Cloud environments.

- NFSv4 only (port 2049)
